cmake_minimum_required(VERSION 3.24)
project(SolarisConsumer LANGUAGES C)
option(SOLARIS_CONSUMER_C_ONLY "Check consumption without enabling C++" OFF)
if(NOT SOLARIS_CONSUMER_C_ONLY)
    enable_language(CXX)
endif()
find_package(Solaris CONFIG REQUIRED)
add_executable(consumer_c main.c)
get_target_property(solaris_kind Solaris::core TYPE)
if(solaris_kind STREQUAL "STATIC_LIBRARY")
    # Force all implementation objects into this pure C host: an engine-only
    # smoke would not pull optional module dependencies from the archive.
    target_link_libraries(consumer_c PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,Solaris::core>")
else()
    target_link_libraries(consumer_c PRIVATE Solaris::core)
endif()
set_target_properties(consumer_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
enable_testing()
add_test(NAME consumer_c COMMAND consumer_c)
if(NOT SOLARIS_CONSUMER_C_ONLY)
    add_executable(consumer_cpp main.cpp)
    target_link_libraries(consumer_cpp PRIVATE Solaris::core)
    set_target_properties(consumer_cpp PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
    add_test(NAME consumer_cpp COMMAND consumer_cpp)
endif()
