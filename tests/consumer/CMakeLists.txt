cmake_minimum_required(VERSION 3.24)
project(SolarisConsumer LANGUAGES C CXX)
find_package(Solaris CONFIG REQUIRED)
add_executable(consumer_c main.c)
target_link_libraries(consumer_c PRIVATE Solaris::core)
set_target_properties(consumer_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
add_executable(consumer_cpp main.cpp)
target_link_libraries(consumer_cpp PRIVATE Solaris::core)
set_target_properties(consumer_cpp PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
enable_testing()
add_test(NAME consumer_c COMMAND consumer_c)
add_test(NAME consumer_cpp COMMAND consumer_cpp)
